Below is a **ready‑to‑use “Sign‑Up” page** written in plain HTML, CSS and a touch of JavaScript for basic client‑side validation.  
Feel free to drop it into a file called `signup.html`, tweak the styles, or plug the form into your favourite backend (PHP, Node, Django, etc.).

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Sign Up – Join Us!</title>
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <!-- Minimal CSS – you can move this to an external stylesheet -->
  <style>
    :root {
      --primary: #4CAF50;          /* Green – feel free to change */
      --bg: #f5f5f5;
      --text: #333;
      --error: #e53935;
    }
    body {
      font-family: Arial, sans-serif;
      background: var(--bg);
      display: flex;
      align-items: center;
      justify-content: center;
      height: 100vh;
      margin: 0;
    }
    .card {
      background: #fff;
      padding: 2rem;
      border-radius: 8px;
      box-shadow: 0 4px 12px rgba(0,0,0,.1);
      width: 320px;
      box-sizing: border-box;
    }
    h2 {
      margin-top: 0;
      text-align: center;
      color: var(--primary);
    }
    label {
      display: block;
      margin: 1rem 0 .5rem;
      font-weight: 600;
    }
    input[type="text"],
    input[type="email"],
    input[type="password"] {
      width: 100%;
      padding: .75rem;
      border: 1px solid #ccc;
      border-radius: 4px;
      font-size: 1rem;
      box-sizing: border-box;
    }
    input[type="text"]:focus,
    input[type="email"]:focus,
    input[type="password"]:focus {
      outline: none;
      border-color: var(--primary);
      box-shadow: 0 0 0 3px rgba(76, 175, 80, .2);
    }
    .error {
      color: var(--error);
      font-size: .85rem;
      margin-top: .25rem;
    }
    button {
      width: 100%;
      padding: .75rem;
      background: var(--primary);
      color: #fff;
      border: none;
      border-radius: 4px;
      font-size: 1rem;
      cursor: pointer;
      margin-top: 1rem;
    }
    button:hover {
      background: #45a049;
    }
  </style>
</head>
<body>

<div class="card">
  <h2>Create an Account</h2>
  <form id="signupForm" novalidate>
    <label for="username">Username</label>
    <input type="text" id="username" name="username" required minlength="3">
    <div class="error" id="usernameError"></div>

    <label for="email">Email</label>
    <input type="email" id="email" name="email" required>
    <div class="error" id="emailError"></div>

    <label for="password">Password</label>
    <input type="password" id="password" name="password" required minlength="6">
    <div class="error" id="passwordError"></div>

    <label for="confirm">Confirm Password</label>
    <input type="password" id="confirm" name="confirm" required>
    <div class="error" id="confirmError"></div>

    <button type="submit">Sign Up</button>
  </form>
</div>

<script>
  // Basic client‑side validation
  document.getElementById('signupForm').addEventListener('submit', function(e) {
    e.preventDefault(); // Prevent actual submission for demo

    // Clear previous errors
    ['usernameError','emailError','passwordError','confirmError'].forEach(id => {
      document.getElementById(id).textContent = '';
    });

    let valid = true;

    const username = document.getElementById('username');
    if (username.value.trim().length < 3) {
      document.getElementById('usernameError').textContent = 'Username must be at least 3 characters.';
      valid = false;
    }

    const email = document.getElementById('email');
    const emailPattern = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;
    if (!emailPattern.test(email.value.trim())) {
      document.getElementById('emailError').textContent = 'Enter a valid email address.';
      valid = false;
    }

    const password = document.getElementById('password');
    if (password.value.length < 6) {
      document.getElementById('passwordError').textContent = 'Password must be at least 6 characters.';
      valid = false;
    }

    const confirm = document.getElementById('confirm');
    if (confirm.value !== password.value) {
      document.getElementById('confirmError').textContent = 'Passwords do not match.';
      valid = false;
    }

    if (valid) {
      // Here you would normally submit the form via AJAX or a normal POST
      alert('All good! Form would now be submitted.');
      // e.g., this.submit();
    }
  });
</script>

</body>
</html>
```

### How to use

1. **Save** the code as `signup.html`.
2. Open the file in a web browser.
3. The form will validate the fields locally and show error messages.
4. Replace the `alert()` in the JavaScript with a real submission (AJAX or native form POST) to your server.

Feel free to adjust the colors, spacing, or add more fields (phone, address, etc.) to fit your project’s style. Happy coding!
